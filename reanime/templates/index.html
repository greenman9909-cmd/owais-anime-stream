Engineered Architecture

Fault-tolerant stream resolution and edge proxying designed for reliability.

3-Server Cluster Failover

Automatic cascading between Sora, Neko, and Zozo edge nodes ensures continuous stream delivery even during regional upstream outages.

Frame-Accurate Subtitles

Synchronized WebVTT and ASS subtitles with custom offset calibration (±10s) rendered directly into HTML5 video cues.

Dual Catalog Sync

Native support for both AniList and MyAnimeList IDs with automatic metadata translation and slug mapping.

HMAC-SHA256 Token Shield

Upstream CDN origins and playback URLs are signed with time-expiring cryptographic tokens to prevent unauthorized hotlinking.

AniSkip Intro & Outro

Direct AniSkip API integration providing timestamp markers to automatically skip openings and credits.

PostMessage API Event Bus

Bidirectional event protocol emitting ready, play, pause, timeupdate, and ended events for smooth integration with host web applications.

import React, { useEffect, useRef } from "react";

interface AnimePlayerProps {
  id: number | string; // AniList ID (e.g. 21) or slug
  episode?: number;
  track?: "sub" | "dub";
  onReady?: (server: string) => void;
  onEnded?: () => void;
  onTimeUpdate?: (currentTime: number, duration: number) => void;
}

export const AnimePlayer: React.FC<AnimePlayerProps> = ({
  id,
  episode = 1,
  track = "sub",
  onReady,
  onEnded,
  onTimeUpdate,
}) => {
  const iframeRef = useRef<HTMLIFrameElement>(null);

  useEffect(() => {
    const handleMessage = (e: MessageEvent) => {
      if (e.data?.source !== "AnimeXOsource_Owais" || e.data?.type !== "yoru:event") return;
      const { event, server, currentTime, duration } = e.data;

      if (event === "ready") onReady?.(server);
      if (event === "timeupdate") onTimeUpdate?.(currentTime, duration);
      if (event === "ended") onEnded?.();
    };

    window.addEventListener("message", handleMessage);
    return () => window.removeEventListener("message", handleMessage);
  }, [onReady, onEnded, onTimeUpdate]);

  return (
    <div style={{ position: "relative", width: "100%", aspectRatio: "16/9", background: "#000" }}>
      <iframe
        ref={iframeRef}
        src={`http://127.0.0.1:8000/embed/ani/${id}/${episode}?track=${track}`}
        style={{ width: "100%", height: "100%", border: "none" }}
        allow="autoplay; fullscreen; picture-in-picture"
        allowFullScreen
        referrerPolicy="no-referrer-when-downgrade"
      />
    </div>
  );
};

// Listen for player events inside your host application
window.addEventListener("message", (event) => {
  // Validate sender
  if (event.data?.source !== "AnimeXOsource_Owais" || event.data?.type !== "yoru:event") return;

  const eventName = event.data.event;
  const payload = event.data;

  switch (eventName) {
    case "ready":
      console.log(`Player initialized on cluster node: ${payload.server}`);
      break;

    case "play":
      console.log("User resumed playback");
      break;

    case "pause":
      console.log("User paused playback");
      break;

    case "timeupdate":
      // Save progress to your user database or localStorage
      const progressPercent = (payload.currentTime / payload.duration) * 100;
      console.log(`Watch progress: ${progressPercent.toFixed(1)}% (${payload.currentTime}s)`);
      break;

    case "skip":
      console.log(`Skipped ${payload.type} to timestamp: ${payload.to}s`);
      break;

    case "serverchange":
      console.log(`User switched server node to: ${payload.server}`);
      break;

    case "failover":
      console.warn(`Node ${payload.from} degraded. Auto-cascaded to: ${payload.to}`);
      break;

    case "ended":
      console.log("Episode finished! Autoplay next episode...");
      // Example: advance episode in state:
      // setEpisode(prev => prev + 1);
      break;

    case "error":
      console.error(`Playback error on ${payload.server}: ${payload.message}`);
      break;
  }
});
