Developer Integration

Embed AnimeXOsource_Owais video streams into your web application in seconds.

<!-- Drop-in AnimeXOsource_Owais Embed Player -->
<iframe
  src="https://owais-anime-stream.onrender.com/embed/one-piece-xamk74/1?track=sub"
  width="100%"
  height="480"
  frameborder="0"
  allowfullscreen
  allow="autoplay; fullscreen; picture-in-picture">
</iframe>
import { searchAnime, getServers, getStreamFromLink, mountHlsPlayer } from "./api-client.js";

// 1. Discover servers
const serverData = await getServers("one-piece-xamk74", 1);
const server = serverData.sub[0] || serverData.dub[0];

// 2. Resolve HLS stream
const stream = await getStreamFromLink(server.dataLink);

// 3. Attach to video element
const video = document.querySelector("#player");
mountHlsPlayer(video, stream.url, stream.subtitles);
video.play();
# 1. Search anime
curl "https://owais-anime-stream.onrender.com/search?q=demon+slayer&limit=5"

# 2. Get playback servers for an episode

# 3. Resolve master.m3u8 directly from embed link

# 4. Health & cluster status
curl "https://owais-anime-stream.onrender.com/health"
window.addEventListener("message", (event) => {
  if (event.data?.source !== "AnimeXOsource_Owais") return;

  switch (event.data.type) {
    case "ready":
      console.log("Player ready on server:", event.data.server);
      break;
    case "timeupdate":
      // Track playback progress
      console.log("Current time:", event.data.currentTime);
      break;
    case "ended":
      console.log("Episode ended, autoplay next!");
      break;
  }
});
